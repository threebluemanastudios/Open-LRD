  New-Item -ItemType Directory -Force $dir | Out-Null

  $small = Join-Path $dir "small-valid.txt"
  [IO.File]::WriteAllText($small,
    "Alice works at Acme.`r`n`r`nShe sees Bob.`r`n",
    (New-Object Text.UTF8Encoding($true)))          # BOM + CRLF

  $large = Join-Path $dir "large.txt"
  $sb = New-Object Text.StringBuilder
  for ($i = 0; $i -lt 140; $i++) {
    [void]$sb.AppendLine(
      "Entity$i works at Company$i in the northern regional office with a " +
      "small project team and a shared annual budget that covers travel and equipment.")
  }
  [IO.File]::WriteAllText($large, $sb.ToString(),
    (New-Object Text.UTF8Encoding($false)))         # > 20,000 characters

  $mixed = Join-Path $dir "mixed.txt"
  [IO.File]::WriteAllText($mixed,
    "Alice works at Acme.`n`nQqq zzz.`n`nBob works at Globex.`n",
    (New-Object Text.UTF8Encoding($false)))

  $binary = Join-Path $dir "binary.bin"

  Get-ChildItem $dir
